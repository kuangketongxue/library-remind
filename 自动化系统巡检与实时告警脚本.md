# 自动化系统巡检与实时告警脚本

> 版本：1.0  
> 创建时间：2026-05-18  
> 脚本类型：PowerShell 持续监控  
> 告警方式：企业微信/飞书机器人 Webhook

---

## 📋 脚本功能概述

本脚本提供以下核心功能：

1. **实时弹窗监控** - 检测异常窗口弹出
2. **可疑进程捕获** - 识别潜在恶意进程
3. **启动项变更告警** - 监控注册表启动项变化
4. **计划任务变动检测** - 跟踪计划任务状态变更
5. **5分钟内告警推送** - 异常即时通知责任人

## 🚀 快速部署

### 前提条件
- PowerShell 5.1 或更高版本
- Windows 10/11 操作系统
- 企业微信/飞书机器人 Webhook URL

```powershell
# 1. 克隆或下载脚本
# 2. 配置 Webhook URL
# 3. 设置执行权限
Set-ExecutionPolicy -ExecutionPolicy RemoteSigned -Scope CurrentUser

# 4. 启动监控
.\auto-monitor.ps1 -Start
```

```powershell
# ============================================
# 🔧 配置区域 - 开始
# ============================================

$Config = @{
    # 飞书机器人 Webhook URL
    FeishuWebhook = "https://open.feishu.cn/open-apis/bot/v2/hook/xxxxxx"
    
    # 企业微信机器人 Webhook URL  
    WechatWebhook = "https://qyapi.weixin.qq.com/cgi-bin/webhook/send?key=xxxxxx"
    
    # 告警接收人（多个用逗号分隔）
    AlertReceivers = @("binlo", "admin")
    
    # 检查间隔（秒）
    CheckInterval = 30
    
    # 告警阈值
    Thresholds = @{
        CpuUsage     = 80    # CPU使用率超过80%告警
        MemoryUsage  = 85    # 内存使用率超过85%告警
        DiskUsage    = 90    # 磁盘使用率超过90%告警
        ResponseTime = 100   # 响应时间超过100ms告警
    }
    
    # 白名单（不告警的进程名）
    WhiteList = @(
        "chrome",
        "WeChatAppEx",
        "TRAE",
        "Code",
        "explorer"
    )
    
    # 日志路径
    LogPath = "$PSScriptRoot\logs\monitor.log"
}

# ============================================
# 🔧 配置区域 - 结束
# ============================================
```

## 📊 监控模块详解

### 4.1 弹窗监控模块

```powershell
function Monitor-PopupWindows {
    <#
    .SYNOPSIS
        监控系统中出现的异常弹窗
    
    .DESCRIPTION
        定期检查所有顶层窗口，识别不在白名单中的弹窗程序
        发现异常立即告警并记录详细信息
    #>
    
    $windows = Get-Process | Where-Object {
        $_.MainWindowHandle -ne 0 -and 
        $_.MainWindowTitle -ne "" -and
        $_.ProcessName -notin $Config.WhiteList
    }
    
    foreach ($window in $windows) {
        $alertMsg = @{
            msg_type = "text"
            content  = @{
                text = @"
🚨 异常弹窗检测

进程名称: $($window.ProcessName)
窗口标题: $($window.MainWindowTitle)
进程ID: $($window.Id)
发现时间: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')

⚠️ 建议立即处理
"@
            }
        }
        
        Send-Alert -Message $alertMsg
        Write-Log -Level "WARNING" -Message "弹窗检测: $($window.ProcessName) - $($window.MainWindowTitle)"
    }
}
```

### 4.2 可疑进程监控

```powershell
function Monitor-SuspiciousProcesses {
    <#
    .SYNOPSIS
        监控可疑进程
    
    .DESCRIPTION
        识别高危关键词相关进程，包括：
        - 弹窗/广告相关进程
        - 挖矿/木马特征进程
        - 异常网络连接进程
    #>
    
    $suspiciousKeywords = @(
        'popup', 'ad', 'promo', 'notify', 'alert',
        'miner', 'trojan', 'malware', 'virus',
        'inject', 'hook', 'keylog'
    )
    
    $processes = Get-Process | Where-Object {
        $procName = $_.ProcessName.ToLower()
        $keywordMatch = $suspiciousKeywords | Where-Object { $procName -match $_ }
        $keywordMatch -and $_.ProcessName -notin $Config.WhiteList
    }
    
    foreach ($proc in $processes) {
        $alertMsg = @{
            msg_type = "text"
            content  = @{
                text = @"
🚨 可疑进程检测

进程名称: $($proc.ProcessName)
进程ID: $($proc.Id)
CPU占用: $([math]::Round($proc.CPU, 2))%
内存占用: $([math]::Round($proc.WorkingSet64 / 1MB, 2)) MB
路径: $($proc.Path)
发现时间: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')

⚠️ 建议立即终止并排查
"@
            }
        }
        
        Send-Alert -Message $alertMsg
        Write-Log -Level "CRITICAL" -Message "可疑进程: $($proc.ProcessName)"
    }
}
```

### 4.3 启动项变更监控

```powershell
function Monitor-StartupChanges {
    <#
    .SYNOPSIS
        监控注册表启动项变更
    
    .DESCRIPTION
        记录启动项基线，检测新增/删除的启动项
        发现未授权变更立即告警
    #>
    
    static $baseline = $null
    
    if (-not $baseline) {
        # 初始化基线
        $baseline = @{
            HKCU = @(Get-ChildItem "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run" | Select-Object -ExpandProperty Name)
            HKLM = @(Get-ChildItem "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run" | Select-Object -ExpandProperty Name)
        }
        Write-Log -Level "INFO" -Message "启动项基线已初始化"
        return
    }
    
    # 检查变更
    $currentHKCU = @(Get-ChildItem "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run" | Select-Object -ExpandProperty Name)
    $currentHKLM = @(Get-ChildItem "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run" | Select-Object -ExpandProperty Name)
    
    $addedHKCU = Compare-Object -ReferenceObject $baseline.HKCU -DifferenceObject $currentHKCU | Where-Object { $_.SideIndicator -eq "=>" }
    $addedHKLM = Compare-Object -ReferenceObject $baseline.HKLM -DifferenceObject $currentHKLM | Where-Object { $_.SideIndicator -eq "=>" }
    
    if ($addedHKCU -or $addedHKLM) {
        $alertMsg = @{
            msg_type = "text"
            content  = @{
                text = @"
🚨 启动项变更检测

新增启动项:
$($addedHKCU | ForEach-Object { "  - HKCU: $($_.InputObject)" }) 
$($addedHKLM | ForEach-Object { "  - HKLM: $($_.InputObject)" })

发现时间: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')

⚠️ 请确认是否为授权变更
"@
            }
        }
        
        Send-Alert -Message $alertMsg
        Write-Log -Level "WARNING" -Message "启动项变更检测: $($addedHKCU.InputObject) $($addedHKLM.InputObject)"
    }
}
```

### 4.4 计划任务监控

```powershell
function Monitor-ScheduledTasks {
    <#
    .SYNOPSIS
        监控计划任务状态变更
    
    .DESCRIPTION
        跟踪计划任务的启用/禁用状态
        检测新增或状态变更的任务
    #>
    
    static $baseline = $null
    
    if (-not $baseline) {
        # 初始化基线
        $baseline = @{
            Tasks = @(Get-ScheduledTask | Where-Object { $_.State -ne "Disabled" } | Select-Object TaskName, State)
        }
        Write-Log -Level "INFO" -Message "计划任务基线已初始化"
        return
    }
    
    # 检查当前状态
    $currentTasks = @(Get-ScheduledTask | Where-Object { $_.State -ne "Disabled" } | Select-Object TaskName, State)
    
    # 检测新增任务
    $diff = Compare-Object -ReferenceObject $baseline.Tasks -DifferenceObject $currentTasks -Property TaskName, State
    
    if ($diff) {
        $alertMsg = @{
            msg_type = "text"
            content  = @{
                text = @"
🚨 计划任务变更检测

变更详情:
$($diff | ForEach-Object { "  - $($_.TaskName): $($_.SideIndicator) " })

发现时间: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')

⚠️ 请确认是否为授权变更
"@
            }
        }
        
        Send-Alert -Message $alertMsg
        Write-Log -Level "WARNING" -Message "计划任务变更: $($diff | Out-String)"
    }
}
```

### 4.5 资源使用率监控

```powershell
function Monitor-ResourceUsage {
    <#
    .SYNOPSIS
        监控系统资源使用率
    
    .DESCRIPTION
        监控CPU、内存、磁盘使用率
        超过阈值时触发告警
    #>
    
    # CPU 使用率
    $cpu = Get-Counter '\Processor(_Total)\% Processor Time' -SampleInterval 1 -MaxSamples 3
    $cpuUsage = [math]::Round(($cpu.CounterSamples | Measure-Object -Property CookedValue -Average).Average, 2)
    
    if ($cpuUsage -gt $Config.Thresholds.CpuUsage) {
        $alertMsg = @{
            msg_type = "text"
            content  = @{
                text = @"
⚠️ CPU使用率告警

当前使用率: $cpuUsage%
阈值: $($Config.Thresholds.CpuUsage)%
检测时间: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')

建议: 检查高负载进程
"@
            }
        }
        
        Send-Alert -Message $alertMsg
        Write-Log -Level "WARNING" -Message "CPU使用率过高: $cpuUsage%"
    }
    
    # 内存使用率
    $os = Get-CimInstance -ClassName Win32_OperatingSystem
    $memUsage = [math]::Round((($os.TotalVisibleMemorySize - $os.FreePhysicalMemory) / $os.TotalVisibleMemorySize) * 100, 2)
    
    if ($memUsage -gt $Config.Thresholds.MemoryUsage) {
        $alertMsg = @{
            msg_type = "text"
            content  = @{
                text = @"
⚠️ 内存使用率告警

当前使用率: $memUsage%
阈值: $($Config.Thresholds.MemoryUsage)%
检测时间: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')

建议: 检查内存占用进程
"@
            }
        }
        
        Send-Alert -Message $alertMsg
        Write-Log -Level "WARNING" -Message "内存使用率过高: $memUsage%"
    }
}
```

### 4.6 告警发送模块

```powershell
function Send-Alert {
    param(
        [Parameter(Mandatory)]
        [hashtable]$Message
    )
    
    # 飞书告警
    try {
        $response = Invoke-RestMethod -Uri $Config.FeishuWebhook -Method Post -Body ($Message | ConvertTo-Json -Depth 5) -ContentType "application/json"
        Write-Log -Level "INFO" -Message "飞书告警发送成功"
    }
    catch {
        Write-Log -Level "ERROR" -Message "飞书告警发送失败: $($_.Exception.Message)"
    }
    
    # 企业微信告警
    try {
        $response = Invoke-RestMethod -Uri $Config.WechatWebhook -Method Post -Body ($Message | ConvertTo-Json -Depth 5) -ContentType "application/json"
        Write-Log -Level "INFO" -Message "企业微信告警发送成功"
    }
    catch {
        Write-Log -Level "ERROR" -Message "企业微信告警发送失败: $($_.Exception.Message)"
    }
}
```

### 4.7 日志模块

```powershell
function Write-Log {
    param(
        [Parameter(Mandatory)]
        [ValidateSet("INFO", "WARNING", "ERROR", "CRITICAL")]
        [string]$Level,
        
        [Parameter(Mandatory)]
        [string]$Message
    )
    
    $timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    $logEntry = "[$timestamp] [$Level] $Message"
    
    # 确保日志目录存在
    $logDir = Split-Path $Config.LogPath -Parent
    if (-not (Test-Path $logDir)) {
        New-Item -ItemType Directory -Path $logDir -Force | Out-Null
    }
    
    # 写入日志
    Add-Content -Path $Config.LogPath -Value $logEntry
    
    # 控制台输出
    switch ($Level) {
        "ERROR" { Write-Host $logEntry -ForegroundColor Red }
        "WARNING" { Write-Host $logEntry -ForegroundColor Yellow }
        "CRITICAL" { Write-Host $logEntry -BackgroundColor Red -ForegroundColor White }
        default { Write-Host $logEntry -ForegroundColor Green }
    }
}
```

## 🎯 主监控循环

```powershell
function Start-Monitoring {
    <#
    .SYNOPSIS
        启动监控系统
    
    .DESCRIPTION
        初始化基线后进入主监控循环
        按配置间隔执行各项检查
    #>
    
    Write-Log -Level "INFO" -Message "监控系统启动"
    Write-Log -Level "INFO" -Message "检查间隔: $($Config.CheckInterval)秒"
    
    # 初始化基线
    Monitor-StartupChanges
    Monitor-ScheduledTasks
    
    # 主循环
    while ($true) {
        try {
            # 并行执行检查
            $jobs = @(
                # 弹窗监控
                { Monitor-PopupWindows },
                
                # 可疑进程监控
                { Monitor-SuspiciousProcesses },
                
                # 启动项监控
                { Monitor-StartupChanges },
                
                # 计划任务监控
                { Monitor-ScheduledTasks },
                
                # 资源监控
                { Monitor-ResourceUsage }
            )
            
            # 执行检查
            foreach ($job in $jobs) {
                & $job
            }
            
            # 等待下次检查
            Start-Sleep -Seconds $Config.CheckInterval
        }
        catch {
            Write-Log -Level "ERROR" -Message "监控循环异常: $($_.Exception.Message)"
            Start-Sleep -Seconds $Config.CheckInterval
        }
    }
}
```
